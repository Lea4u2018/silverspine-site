Silver Spine Studio — Firestore rules for Reviews
Paste into: Firebase Console → Firestore Database → Rules → Publish

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /reviews/{reviewId} {
      // Public page only queries approved==true.
      // Admin needs to read pending too.
      allow read: if true;

      // Anyone may submit a pending review (not pre-approved).
      allow create: if
        request.resource.data.approved == false
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && request.resource.data.name.size() <= 80
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 2000
        && request.resource.data.rating is number
        && request.resource.data.rating >= 0
        && request.resource.data.rating <= 5;

      // Approve / unpublish / decline (soft delete via rejected).
      allow update: if
        request.resource.data.name == resource.data.name
        && request.resource.data.text == resource.data.text
        && request.resource.data.rating == resource.data.rating;

      allow delete: if false;
    }
  }
}
